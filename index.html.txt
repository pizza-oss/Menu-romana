  <script>
    const WHATSAPP_NUMBER = "529624037244";

    const SIZES = [
      { name: "Chica", price: 85 },
      { name: "Mediana", price: 135 },
      { name: "Grande", price: 165 },
      { name: "Familiar", price: 195 },
      { name: "Jumbo", price: 240 }
    ];

    const CLASSIC_PIZZAS = [
      { name: "Hawaiana", desc: "Queso mozzarella, jamón, tocineta, piña" },
      { name: "Pepperoni / Salami", desc: "Queso mozzarella, pepperoni o salami" },
      { name: "Carnes Frías", desc: "Queso mozzarella, chorizo, salchicha, jamón y tocino" },
      { name: "Champeroni", desc: "Queso mozzarella, pepperoni y champiñones" },
      { name: "Ranchera", desc: "Queso mozzarella, chorizo, frijol, rajas" },
      { name: "Vegetariana", desc: "Queso mozzarella, cebolla, champiñones, pimentón, pollo, elote, tocino" },
      { name: "Jamón / Champiñón", desc: "Queso mozzarella, jamón y champiñones" }
    ];

    const SPECIAL_PIZZAS = [
      { name: "Atún", desc: "Queso mozzarella, atún, queso crema, parmesano" },
      { name: "Quesos", desc: "Queso mozzarella, queso parmesano y queso crema" },
      { name: "Mexicana", desc: "Jamón, queso mozzarella, chorizo, rajas, cebolla y aguacate" },
      { name: "Pastor", desc: "Queso mozzarella, carne al pastor, piña" },
      { name: "Carbonara", desc: "Tocino, parmesano, cebolla y queso mozzarella" },
      { name: "Arrachera", desc: "Queso mozzarella, carne de arrachera, tocino, chistorra" }
    ];

    let cart = [];
    let currentPizza = null;
    let selectedSizeIndex = 2; // Grande por defecto

    function renderMenu(list, containerId) {
      const container = document.getElementById(containerId);
      list.forEach((pizza) => {
        const card = document.createElement("div");
        card.className = "pizza-card";
        card.innerHTML = `
          <div class="pizza-info">
            <h3>${pizza.name}</h3>
            <p>${pizza.desc}</p>
          </div>
          <button class="btn-select" onclick='openSizeModal(${JSON.stringify(pizza)})'>+ Pedir</button>
        `;
        container.appendChild(card);
      });
    }

    function openSizeModal(pizza) {
      currentPizza = pizza;
      selectedSizeIndex = 2;
      document.getElementById("modal-pizza-name").textContent = pizza.name;
      document.getElementById("modal-pizza-desc").textContent = pizza.desc;
      renderSizeOptions();
      document.getElementById("size-modal").classList.add("active");
    }

    function closeSizeModal() {
      document.getElementById("size-modal").classList.remove("active");
    }

    function renderSizeOptions() {
      const container = document.getElementById("size-options");
      container.innerHTML = "";
      SIZES.forEach((size, index) => {
        const div = document.createElement("div");
        div.className = "size-option" + (index === selectedSizeIndex ? " selected" : "");
        div.innerHTML = `<span>${size.name}</span><strong>$${size.price}</strong>`;
        div.onclick = () => {
          selectedSizeIndex = index;
          renderSizeOptions();
        };
        container.appendChild(div);
      });
    }

    function confirmAddPizza() {
      const size = SIZES[selectedSizeIndex];
      cart.push({
        pizza: currentPizza.name,
        size: size.name,
        price: size.price
      });
      updateCartBar();
      closeSizeModal();
    }

    function updateCartBar() {
      const bar = document.getElementById("cart-bar");
      if (cart.length === 0) {
        bar.classList.add("hidden");
        return;
      }
      bar.classList.remove("hidden");
      const total = cart.reduce((sum, item) => sum + item.price, 0);
      document.getElementById("cart-count").textContent =
        cart.length === 1 ? "1 pizza seleccionada" : `${cart.length} pizzas seleccionadas`;
      document.getElementById("cart-total").textContent = `$${total} MXN`;
    }

    function openCheckout() {
      const list = document.getElementById("cart-items-list");
      list.innerHTML = "";
      cart.forEach((item, idx) => {
        const row = document.createElement("div");
        row.className = "cart-item";
        row.innerHTML = `
          <div>
            <strong>1x ${item.pizza}</strong> (${item.size})
          </div>
          <div>
            $${item.price}
            <button class="btn-remove" onclick="removeItem(${idx})">Quitar</button>
          </div>
        `;
        list.appendChild(row);
      });
      document.getElementById("checkout-modal").classList.add("active");
    }

    function closeCheckout() {
      document.getElementById("checkout-modal").classList.remove("active");
    }

    function removeItem(index) {
      cart.splice(index, 1);
      updateCartBar();
      if (cart.length === 0) {
        closeCheckout();
      } else {
        openCheckout();
      }
    }

    function sendWhatsApp() {
      if (cart.length === 0) return;
      const zone = document.getElementById("delivery-zone").value;
      const address = document.getElementById("customer-address").value.trim();
      const total = cart.reduce((sum, item) => sum + item.price, 0);

      let message = "🍕 *Nuevo Pedido - Pizzería La Romana*\n\n";
      cart.forEach((item, i) => {
        message += `${i + 1}. *${item.pizza}* (${item.size}) - $${item.price}\n`;
      });
      message += `\n💰 *Total:* $${total} MXN`;
      message += `\n📍 *Zona:* ${zone}`;
      if (address) {
        message += `\n🏠 *Dirección / Nombre:* ${address}`;
      }

      const url = `https://wa.me/${WHATSAPP_NUMBER}?text=${encodeURIComponent(message)}`;
      window.open(url, "_blank");
    }

    renderMenu(CLASSIC_PIZZAS, "classic-menu");
    renderMenu(SPECIAL_PIZZAS, "special-menu");
  </script>
